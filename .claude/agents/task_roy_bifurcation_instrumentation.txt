**New Code (Sylvia's sigmoid recommendation):**
```typescript
private getSystemMultiplier(
  thresholdName: string,
  currentMonth: number
): number {
  // Base multipliers (full strength late-game)
  const baseMultipliers: Record<string, number> = {
    'environmental': 1.5,
    'social': 2.5,
    'economic': 2.5,
    'governance': 2.0,
    'flourishing': 2.0,
    'technology': 2.0,
  };

  const baseMultiplier = baseMultipliers[thresholdName] ?? 2.0;

  // Time-based scaling (smooth sigmoid ramp)
  // Early simulation (months 0-60): ~0.5× multipliers (reduced amplification)
  // Mid simulation (months 60-180): Smooth transition
  // Late simulation (months 180+): 1.0× multipliers (full amplification)
  //
  // Research basis: Arumugam et al. (2024 Ecology) - Rate-dependent transitions
  // Fast scenarios (20-year simulation) show different dynamics than slow drift
  const centerMonth = 120; // Midpoint of transition
  const transitionWidth = 60; // ±60 months (10 years total transition window)
  const progress = (currentMonth - centerMonth) / transitionWidth;
  const timeScaling = 0.5 + 0.5 / (1 + Math.exp(-progress));
  // Sigmoid ramps from ~0.5 (month 0) to ~1.0 (month 240)

  return baseMultiplier * timeScaling;
}
```

// Export bifurcation metrics for Priya validation (Nov 13, 2025)
function exportBifurcationMetrics(
  finalState: GameState,
  seed: number,
  outputDir: string
): void {
  if (!finalState.bifurcationState.metrics) {
    console.warn(`⚠️ Seed ${seed}: No bifurcation metrics available`);
    return;
  }

  const metrics = finalState.bifurcationState.metrics;
  const bifState = finalState.bifurcationState;

  // Build per-domain bifurcation data
  const domains = {
    environmental: {
      occurred: bifState.environmentalCollapseThreshold.crossed,
      month: bifState.environmentalCollapseThreshold.crossedAt,
      type: 'fold_catastrophe',
      threshold: bifState.environmentalCollapseThreshold.location,
    },
    social: {
      occurred: bifState.socialBreakdownThreshold.crossed,
      month: bifState.socialBreakdownThreshold.crossedAt,
      type: 'hopf_bifurcation',
      threshold: bifState.socialBreakdownThreshold.location,
    },
    economic: {
      occurred: bifState.economicCollapseThreshold.crossed,
      month: bifState.economicCollapseThreshold.crossedAt,
      type: 'cascade_amplification',
      threshold: bifState.economicCollapseThreshold.location,
    },
    governance: {
      occurred: bifState.governanceFailureThreshold.crossed,
      month: bifState.governanceFailureThreshold.crossedAt,
      type: 'feedback_loop',
      threshold: bifState.governanceFailureThreshold.location,
    },
    technology: {
      occurred: bifState.technologyBreakthroughThreshold.crossed,
      month: bifState.technologyBreakthroughThreshold.crossedAt,
      type: 'innovation_cascade',
      threshold: bifState.technologyBreakthroughThreshold.location,
    },
    flourishing: {
      occurred: bifState.flourishingThreshold.crossed,
      month: bifState.flourishingThreshold.crossedAt,
      type: 'positive_feedback',
      threshold: bifState.flourishingThreshold.location,
    },
  };

  // Extract time series data
  const amplificationTimeSeries = metrics.amplificationTimeSeries.map(entry => ({
    month: entry.month,
    amplification: entry.amplification,
    distanceToNearest: entry.distanceToNearest,
    nearestSystem: entry.nearestSystem,
  }));

  // Build output JSON
  const output = {
    seed,
    months: finalState.currentMonth,
    outcome: classifyOutcome(finalState), // Use existing outcome classifier
    finalPopulation: finalState.humanPopulationSystem.population,
    finalQOL: finalState.qualityOfLifeSystems.aggregate.overall,
    bifurcations: domains,
    maxVarianceAmplification: metrics.maxVarianceAmplification,
    avgDistanceToThresholds: metrics.avgDistanceToThresholds,
    amplificationTimeSeries,
    regimeShiftEvents: metrics.regimeShiftEvents,
  };

  // Write to file
  const filename = path.join(outputDir, `bifurcation_metrics_seed${seed}.json`);
  fs.writeFileSync(filename, JSON.stringify(output, null, 2), 'utf8');
  console.log(`📊 Bifurcation metrics exported: ${filename}`);
}

// Call after each run:
exportBifurcationMetrics(finalState, seed, outputDir);
```

**Helper function (add classifyOutcome if not already exported):**

// Import from outcome classifier or implement simple version:
function classifyOutcome(state: GameState): string {
  const pop = state.humanPopulationSystem.population;
  const qol = state.qualityOfLifeSystems.aggregate.overall;

  if (pop < 1e8) return 'EXTINCTION';
  if (pop < 1e9) return 'COLLAPSE';
  if (qol < 20) return 'DYSTOPIA';
  if (qol < 40) return 'STATUS_QUO';
  if (qol < 60) return 'PROGRESS';
  if (qol < 80) return 'FLOURISHING';
  return 'UTOPIA';
}
```
